// The "teobeanie/test" module ("teob/test", its old name, too): tests that run in the Studio, on the add-on's own simulator.
import type { Action, AnyHeuristic, EntityId } from "./index.js";

export interface Sim {
  /** Virtual clock, ms epoch. */
  readonly now: number;
  /** Set an entity's state as HA reports it; delivers a signal when the heuristic watches it. */
  set(entity: EntityId, state: string, attributes?: Record<string, unknown>): Sim;
  /** Advance the virtual clock, firing due timers in order. */
  advance(ms: number): Sim;
  /** Deliver the start input (boot reconcile). */
  start(): Sim;
  trigger(id: string, data?: Record<string, unknown>): Sim;
  /** Calls that would have reached HA: "light.turn_on light.hallway", "notify: text". */
  calls(): string[];
  actions(): Array<{ at: number; action: Action; disposition: "requested" | "suppressed" | "refused"; reason?: string }>;
  /** Every decision's reason, in order. */
  whys(): string[];
  readonly journal: Array<{ at: number; event: { tag: string; [field: string]: unknown } }>;
}

/**
 * Run a rule or heuristic on a scripted house and a virtual clock.
 * t0: start time (ms or ISO string, default Monday 2026-01-05 12:00 UTC).
 */
export declare function simulate(
  heuristic: AnyHeuristic,
  params?: unknown,
  world?: Partial<Record<EntityId, string>>,
  opts?: { t0?: number | string; mode?: "active" | "shadow" },
): Sim;

export declare function test(name: string, fn: () => void): void;
/** The same as test(). */
export declare function it(name: string, fn: () => void): void;
/** Groups tests: their names start with this one's. */
export declare function describe(name: string, fn: () => void): void;

export interface Matchers {
  toBe(expected: unknown): void;
  toEqual(expected: unknown): void;
  toHaveLength(n: number): void;
  toContain(item: unknown): void;
  toMatch(pattern: RegExp | string): void;
  toBeTruthy(): void;
  toBeFalsy(): void;
  toThrow(pattern?: RegExp | string): void;
  readonly not: Matchers;
}
export declare function expect(actual: unknown): Matchers;
